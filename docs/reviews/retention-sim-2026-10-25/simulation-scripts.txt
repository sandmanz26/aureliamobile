# Retention model for the persona simulation (J039).
# Probabilities are judgement calls by the simulated personas, anchored to
# public Health & Fitness benchmarks (D1 ~20-27%, D7 ~8-13%, D30 ~3-8%).
# Stage: act = finishes a first session in visit 1; r1 = D1 | act;
# r7 = D7 | D1; r30 = D30 | D7. Each fix multiplies the stages it touches.
import json, itertools

P = {  # weight, S1 base: act, r1, r7, r30 (as observed on the build, persistence assumed)
 'maya': dict(w=.25, act=.60, r1=.30, r7=.32, r30=.40),
 'raka': dict(w=.20, act=.38, r1=.18, r7=.25, r30=.35),
 'sari': dict(w=.20, act=.45, r1=.26, r7=.30, r30=.38),
 'leo':  dict(w=.15, act=.25, r1=.28, r7=.22, r30=.25),
 'dewi': dict(w=.10, act=.52, r1=.30, r7=.40, r30=.50),
 'tom':  dict(w=.10, act=.45, r1=.26, r7=.36, r30=.46),
}
# Fix -> {persona: (act, r1, r7, r30) multipliers}; '*' = default for all.
F = {
 'R1 One first message, in their words': {'*': (1.06, 1.03, 1, 1), 'maya': (1.10, 1.05, 1, 1), 'tom': (1.10, 1.05, 1, 1), 'sari': (1.08, 1.03, 1, 1)},
 'R2 A fresh first thread, no early paywall': {'*': (1.15, 1.10, 1.02, 1), 'maya': (1.02, 1.02, 1, 1), 'raka': (1.20, 1.18, 1.03, 1), 'dewi': (1.18, 1.15, 1.03, 1), 'tom': (1.20, 1.15, 1.03, 1)},
 'R3 Build from the goal': {'*': (1.10, 1.10, 1.04, 1.02), 'raka': (1.15, 1.18, 1.05, 1.02), 'dewi': (1.12, 1.15, 1.05, 1.03), 'leo': (1.15, 1.10, 1.03, 1)},
 'R4 A Play that plays': {'*': (1.04, 1, 1, 1), 'leo': (1.45, 1.06, 1, 1), 'sari': (1.08, 1, 1, 1), 'maya': (1.06, 1.03, 1, 1)},
 'R5 Honest day 0 + one-tap check-in': {'*': (1.00, 1.12, 1.08, 1.05), 'raka': (1.0, 1.25, 1.12, 1.05), 'tom': (1.0, 1.22, 1.10, 1.05), 'dewi': (1.0, 1.18, 1.10, 1.05)},
 'R6 Chapter 2 reminder': {'*': (1, 1.25, 1.22, 1.10), 'leo': (1, 1.30, 1.30, 1.12), 'sari': (1, 1.30, 1.25, 1.10), 'dewi': (1, 1.10, 1.08, 1.05)},
 'R7 Personal Home on return': {'*': (1, 1.12, 1.10, 1.05), 'leo': (1, 1.15, 1.12, 1.05)},
 'R8 Real progress from check-ins': {'*': (1, 1.03, 1.15, 1.20), 'raka': (1, 1.05, 1.20, 1.25), 'dewi': (1, 1.03, 1.18, 1.25), 'tom': (1, 1.03, 1.18, 1.25)},
 'R9 Only their own things': {'*': (1.02, 1.04, 1.02, 1), 'raka': (1.05, 1.08, 1.03, 1), 'tom': (1.05, 1.08, 1.02, 1)},
 'R10 Invite a second goal': {'*': (1, 1, 1.02, 1.15), 'dewi': (1, 1, 1.03, 1.22), 'maya': (1, 1, 1.02, 1.18)},
}
SIMPLIFY = {'*': (1.04, 1.03, 1.05, 1.06), 'leo': (1.06, 1.04, 1.06, 1.06), 'sari': (1.06, 1.03, 1.05, 1.05)}

def mult(table, who):
  return table.get(who, table['*'])

DAMP = .45  # fixes overlap (two cues are not twice one cue): only part of the compounded lift lands

def run(fixes, simplify=False, damp=DAMP):
  out = {}
  for who, b in P.items():
    base = [b['act'], b['r1'], b['r7'], b['r30']]
    prod = [1, 1, 1, 1]
    for t in [F[f] for f in fixes] + ([SIMPLIFY] if simplify else []):
      prod = [x * m for x, m in zip(prod, mult(t, who))]
    s = [min(x * (1 + damp * (m - 1)), .9) for x, m in zip(base, prod)]
    act, r1, r7, r30 = s
    d1 = act * r1; d7 = d1 * r7; d30 = d7 * r30
    out[who] = dict(act=act, d1=d1, d7=d7, d30=d30)
  w = {k: sum(P[p]['w'] * out[p][k] for p in P) for k in ('act', 'd1', 'd7', 'd30')}
  return out, w

ALL = list(F)
res = {}
for name, fixes, simp in [('S1', [], False), ('S2', ALL, False), ('S3', ALL, True)]:
  per, w = run(fixes, simp)
  res[name] = dict(per=per, w=w, lo=run(fixes, simp, .25)[1], hi=run(fixes, simp, .65)[1])
# The S1 base rates are the least certain number here: +/-30% on retention, +/-15% on activation
res['S1']['lo'] = {k: v * (.85 if k == 'act' else .7) for k, v in res['S1']['w'].items()}
res['S1']['hi'] = {k: v * (1.15 if k == 'act' else 1.3) for k, v in res['S1']['w'].items()}
# Shapley value of each fix: its average marginal contribution over every order
# the nine could ship in. They sum exactly to S2 - S1.
from math import factorial
n = len(ALL); cache = {}
def val(sub):
  key = frozenset(sub)
  if key not in cache: cache[key] = run(list(sub))[1]
  return cache[key]
res['fix'] = {}
for f in ALL:
  others = [g for g in ALL if g != f]; acc = dict(act=0, d1=0, d7=0, d30=0)
  for k in range(len(others) + 1):
    for sub in itertools.combinations(others, k):
      wgt = factorial(k) * factorial(n - k - 1) / factorial(n)
      a = val(sub + (f,)); b = val(sub)
      for m in acc: acc[m] += wgt * (a[m] - b[m])
  res['fix'][f] = acc
json.dump(res, open('ret/model.json', 'w'), indent=1)
for s_ in ('S1', 'S2', 'S3'):
  w = res[s_]['w']; print(s_, ' '.join(f"{k}={w[k]*100:.1f}%" for k in w), '| lo', ' '.join(f"{res[s_]['lo'][k]*100:.1f}" for k in w), '| hi', ' '.join(f"{res[s_]['hi'][k]*100:.1f}" for k in w))
for s_ in ('S1','S2','S3'):
  print(s_, {p: f"{v['act']*100:.0f}|{v['d1']*100:.0f}/{v['d7']*100:.1f}/{v['d30']*100:.1f}" for p, v in res[s_]['per'].items()})
for f, v in sorted(res['fix'].items(), key=lambda x: -x[1]['d7']):
  print(f"{f:34s} D1 +{v['d1']*100:.1f} D7 +{v['d7']*100:.2f} D30 +{v['d30']*100:.2f}")
so = run([], True)[1]; res['SIMP'] = dict(w=so)
first5 = ['R1 One first message, in their words', 'R2 A fresh first thread, no early paywall', 'R3 Build from the goal', 'R5 Honest day 0 + one-tap check-in', 'R6 Chapter 2 reminder']
res['TOP'] = dict(w=run(first5)[1], fixes=first5)
json.dump(res, open('ret/model.json', 'w'), indent=1)
print('SIMP only', {k: round(v*100, 1) for k, v in so.items()}); print('TOP5', {k: round(v*100, 1) for k, v in res['TOP']['w'].items()})

import json
M = json.load(open('ret/model.json'))
C = {'S1': '#D9B08C', 'S2': '#D9822B', 'S3': '#6B3608'}
NAME = {'S1': 'Current v4', 'S2': '+ 9 retention fixes', 'S3': '+ fixes + simplification'}
INK, MUTED, RULE = '#1B1006', '#7A6A58', '#E8DFD3'
def T(x, y, s, size=10, fill=MUTED, anchor='start', weight=400):
  return f'<text x="{x:.1f}" y="{y:.1f}" font-size="{size}" fill="{fill}" text-anchor="{anchor}" font-weight="{weight}" font-family="Mulish">{s}</text>'

def curve():
  W, H, l, r, t, b = 640, 300, 44, 150, 16, 34
  xs = {'d1': l + 20, 'd7': l + (W - l - r) * .42, 'd30': W - r - 10}
  ymax = .40
  y = lambda v: t + (H - t - b) * (1 - v / ymax)
  o = [f'<svg viewBox="0 0 {W} {H}" xmlns="http://www.w3.org/2000/svg" role="img" aria-label="Predicted retention D1, D7, D30 by scenario">']
  for g in (0, .1, .2, .3, .4):
    o.append(f'<line x1="{l}" x2="{W-r+20}" y1="{y(g):.1f}" y2="{y(g):.1f}" stroke="{RULE}" stroke-width="1"/>' + T(l - 8, y(g) + 3.5, f'{int(g*100)}%', 9, anchor='end'))
  band = {'d1': (.20, .27), 'd7': (.08, .13), 'd30': (.03, .08)}
  pts_hi = ' '.join(f'{xs[k]:.1f},{y(band[k][1]):.1f}' for k in xs); pts_lo = ' '.join(f'{xs[k]:.1f},{y(band[k][0]):.1f}' for k in reversed(list(xs)))
  o.append(f'<polygon points="{pts_hi} {pts_lo}" fill="#EFE6DA" opacity=".9"/>')

  for k, lab in (('d1', 'Day 1'), ('d7', 'Day 7'), ('d30', 'Day 30')):
    o.append(T(xs[k], H - 12, lab, 10, INK, 'middle', 600))
  for s in ('S1', 'S2', 'S3'):
    w, lo, hi = M[s]['w'], M[s]['lo'], M[s]['hi']
    pts = ' '.join(f'{xs[k]:.1f},{y(w[k]):.1f}' for k in xs)

      o.append(f'<circle cx="{xs[k]:.1f}" cy="{y(w[k]):.1f}" r="4.2" fill="{C[s]}" stroke="#fff" stroke-width="2"/>')
    dy = {'S1': 4, 'S2': -2, 'S3': -10}[s]
    o.append(T(xs['d30'] + 12, y(w['d30']) + dy if s != 'S1' else y(w['d30']) + 14, f'{NAME[s]} · {w["d30"]*100:.0f}%', 9.5, INK, weight=600))
  for s, dx in (('S1', 10), ('S3', 10)):
    w = M[s]['w']; o.append(T(xs['d1'] + dx, y(w['d1']) + (14 if s == 'S1' else -8), f'{w["d1"]*100:.0f}%', 9.5, INK, weight=700))

    w = M[s]['w']; o.append(T(xs['d7'], y(w['d7']) - 9 if s != 'S1' else y(w['d7']) + 16, f'{w["d7"]*100:.0f}%', 9.5, INK, 'middle', 700))

def fixes():

  W, row, l = 640, 30, 210; H = 34 + row * len(items)
